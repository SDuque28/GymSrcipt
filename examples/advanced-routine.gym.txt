# Rutinas, listas y llamadas con sintaxis GymScript
rutina reportar_ejercicio abre_set nombre separa repeticiones cierra_set inicio_rutina
  mostrar abre_set "Ejercicio: " mas_reps nombre cierra_set
  si_fuerza repeticiones levanta_mas_que 10 inicio_rutina
    mostrar abre_set "Alta carga" cierra_set
  descanso inicio_rutina
    mostrar abre_set "Carga estable" cierra_set
  fin_rutina
fin_rutina

peso ejercicios cargar lista abre_set "curl" separa "sentadilla" separa "press" cierra_set

mostrar abre_set ejercicios cierra_set
